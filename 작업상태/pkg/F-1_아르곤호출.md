# -*- coding: utf-8 -*-
"""아르곤_호출.py - agy(Antigravity CLI)로 Gemini 4 Argon을 부르는 공용 도우미 (2026-10-21)

쓰는 법 (3줄)
 1) 명령줄: py -3 -I 아르곤_호출.py --prompt "질문" --add-dir "D:\\폴더1" --add-dir "D:\\폴더2" --out "D:\\결과\\답.txt"
            (프롬프트가 길면 --prompt-file 파일.txt, 답을 화면에도 보려면 --echo, 환경 점검만은 --check)
 2) 종료코드: 0 성공 / 2 사용법·agy 없음·모델 없음 / 3 한도(429) = 이 계정은 멈추고 다른 계정·경로로 전환 / 4 시간초과 / 5 응답 없음·실패
 3) 파이썬: sys.path.insert(0, r"C:\\Users\\rnjsg\\Downloads\\코덱스 , 클코 파일\\scripts"); from 아르곤_호출 import call
            r = call("질문", add_dirs=[폴더...], out="답.txt"); r["ok"], r["text"], r["code"], r["out"]

하는 일: 실제 agy 경로 찾기 -> 모델 확인(기본 gemini-4-argon-high, 없으면 medium) -> agy -p ... --model M --add-dir D...
        --dangerously-skip-permissions --print-timeout Ns 로 호출 -> 답을 파일(+ .meta.json)로 저장 -> 429면 종료코드 3.
권한: --dangerously-skip-permissions 는 agy 공식 옵션이고 사용자가 허용한 방식이다(Downloads AGENTS.md). 모든 도구 권한을 묻지 않고 자동 승인한다.
        그래서 다룰 폴더는 --add-dir 로 미리 알려 준다. 허용 창을 기다리다 멈추는 일이 없어야 하는 자동화용이다.
동시 실행: agy는 무거워서 이 도우미는 동시에 3개까지만 돌린다(남은 호출은 자리가 날 때까지 기다림, 기본 15분).
경로 구분: 이 도우미는 agy 아르곤(무료 경로, 날짜 제한 없음)이다. Vertex33 아르곤과 다르다 - 영수증·보고에 경로를 구분해 적을 것.
비밀: 키·토큰을 읽지도 쓰지도 않는다. 프롬프트·답은 로컬 파일에만 남는다.
"""
import argparse
import datetime as dt
import importlib.util
import json
import os
import re
import shutil
import subprocess
import sys
import time

WORK_ROOT = r"D:\작업_대용량\아르곤_호출"
DEFAULT_MODEL = "gemini-4-argon-high"
FALLBACK_MODEL = "gemini-4-argon-medium"
MAX_PARALLEL = 3
LONG_PROMPT_CHARS = 7000          # 이보다 길면 파일로 내려놓고 짧은 지시로 부른다(윈도우 명령줄 길이 한도 때문)
MODELS_TTL_SEC = 6 * 3600
USAGE_LEDGER = (r"C:\Users\rnjsg\Downloads\코덱스 , 클코 파일\project_runs\코덱스편집_오퍼스급_20261016"
                r"\무료사용량\gemini_usage.py")

EXIT_OK, EXIT_USAGE, EXIT_LIMIT, EXIT_TIMEOUT, EXIT_FAIL = 0, 2, 3, 4, 5

# 실제 agy 위치 후보: %LOCALAPPDATA% (Claude 앱 안에서는 패키지 쪽으로 바뀌어 보일 수 있음) -> 실제 AppData -> 앱 밖 패키지 경로.
_AGY_CANDIDATES = [
    os.environ.get("AGY_EXE", ""),
    os.path.join(os.environ.get("LOCALAPPDATA", ""), "agy", "bin", "agy.EXE"),
    r"C:\Users\rnjsg\AppData\Local\agy\bin\agy.EXE",
    r"C:\Users\rnjsg\AppData\Local\Packages\Claude_pzs8sxrjxfjjc\LocalCache\Local\agy\bin\agy.EXE",
]
_LIMIT_RE = re.compile(r"\b429\b|resource[_ ]exhausted|quota|rate[ -]?limit|too many requests|usage limit|exhausted your capacity"
                       r"|resets in \d+h|AGY_ERROR",  # 10-21 실측: Individual quota reached ... Resets in 165h57m18s. AGY_ERROR
                       re.I)
_NOMODEL_RE = re.compile(r"(unknown|invalid|unsupported|not found|no such)[^\n]{0,40}model|model[^\n]{0,40}(not found|unavailable|unsupported|invalid)",
                         re.I)
_NOFLAGS = getattr(subprocess, "CREATE_NO_WINDOW", 0)


def find_agy():
    for c in _AGY_CANDIDATES:
        if c and os.path.isfile(c):
            return c
    return shutil.which("agy")


def _stamp():
    return dt.datetime.now().strftime("%Y%m%d_%H%M%S") + "_%d" % os.getpid()


def list_models(agy, refresh=False):
    """agy models 결과의 모델 ID 목록. 6시간 캐시. 못 읽으면 빈 목록(그러면 요청한 모델을 그대로 믿고 부른다)."""
    cache = os.path.join(WORK_ROOT, "models_cache.json")
    if not refresh:
        try:
            d = json.load(open(cache, encoding="utf-8"))
            if time.time() - d.get("at", 0) < MODELS_TTL_SEC and d.get("ids"):
                return d["ids"]
        except (OSError, ValueError):
            pass
    try:
        r = subprocess.run([agy, "models"], capture_output=True, text=True, encoding="utf-8", errors="replace",
                           timeout=90, stdin=subprocess.DEVNULL, creationflags=_NOFLAGS)
    except (OSError, subprocess.TimeoutExpired):
        return []
    ids = [ln.split("\t")[0].strip() for ln in r.stdout.splitlines() if "\t" in ln]
    ids = [i for i in ids if re.fullmatch(r"[A-Za-z0-9._-]+", i)]
    if ids:
        try:
            os.makedirs(WORK_ROOT, exist_ok=True)
            json.dump({"at": time.time(), "ids": ids}, open(cache, "w", encoding="utf-8"), ensure_ascii=False)
        except OSError:
            pass
    return ids


def resolve_model(agy, want):
    ids = list_models(agy)
    if not ids or want in ids:
        return want, ""
    if want == DEFAULT_MODEL and FALLBACK_MODEL in ids:
        return FALLBACK_MODEL, "%s 없음 -> %s 사용" % (want, FALLBACK_MODEL)
    raise ValueError("모델 없음: %s (agy models: %s)" % (want, ", ".join(ids[:20])))


def _pid_alive(pid):
    try:
        import psutil
        return psutil.pid_exists(pid)
    except ImportError:
        try:
            r = subprocess.run(["tasklist", "/FI", "PID eq %d" % pid, "/NH"], capture_output=True, text=True,
                               timeout=20, creationflags=_NOFLAGS)
            return str(pid) in r.stdout
        except (OSError, subprocess.TimeoutExpired):
            return True


class _Slot:
    """동시에 MAX_PARALLEL개까지만 agy를 돌리는 잠금 파일 자리."""

    def __init__(self, wait_sec, stale_sec):
        self.dir = os.path.join(WORK_ROOT, "slots")
        self.wait_sec, self.stale_sec, self.path = wait_sec, stale_sec, None

    def __enter__(self):
        os.makedirs(self.dir, exist_ok=True)
        end = time.time() + self.wait_sec
        while True:
            for i in range(MAX_PARALLEL):
                p = os.path.join(self.dir, "slot%d.lock" % i)
                try:
                    fd = os.open(p, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
                    os.write(fd, str(os.getpid()).encode())
                    os.close(fd)
                    self.path = p
                    return self
                except FileExistsError:
                    try:
                        pid = int(open(p).read().strip() or 0)
                        old = time.time() - os.path.getmtime(p) > self.stale_sec
                        if old or (pid and not _pid_alive(pid)):
                            os.remove(p)
                    except (OSError, ValueError):
                        pass
                except OSError:
                    pass
            if time.time() > end:
                raise TimeoutError("agy 자리 대기 시간 초과(동시 %d개 한도)" % MAX_PARALLEL)
            time.sleep(3)

    def __exit__(self, *a):
        if self.path:
            try:
                os.remove(self.path)
            except OSError:
                pass


def _ledger(model, ok):
    """기존 무료 사용량 장부(있으면)에 1회 호출을 적는다. 실패해도 호출 결과에는 영향 없음."""
    try:
        spec = importlib.util.spec_from_file_location("common_gemini_usage", USAGE_LEDGER)
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        m.record(tool=os.path.basename(sys.argv[0]) or "아르곤_호출", route="ANTIGRAVITY", model=model,
                 at=m.iso(m.now()), account=os.environ.get("GEMINI_USAGE_ACCOUNT"), project=None,
                 operation="cli_invocation", request_count=1, cost_class="SUBSCRIPTION_INCLUDED_USER_CONFIRMED",
                 result="CLI_EXIT_0_UNVERIFIED_CONTENT" if ok else "CLI_FAILED_OR_UNKNOWN")
    except Exception:
        pass


def _run_agy(cmd, timeout):
    """agy를 띄우고 시간이 넘으면 이 호출의 프로세스 트리만 끈다. (returncode, stdout, stderr, timed_out)"""
    p = subprocess.Popen(cmd, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                         text=True, encoding="utf-8", errors="replace", creationflags=_NOFLAGS)
    try:
        out, err = p.communicate(timeout=timeout)
        return p.returncode, out or "", err or "", False
    except subprocess.TimeoutExpired:
        subprocess.run(["taskkill", "/PID", str(p.pid), "/T", "/F"], capture_output=True, creationflags=_NOFLAGS)
        try:
            out, err = p.communicate(timeout=20)
        except Exception:
            out, err = "", ""
        return p.returncode if p.returncode is not None else -1, out or "", err or "", True


def call(prompt, add_dirs=None, out=None, model=DEFAULT_MODEL, print_timeout=600, slot_wait=900, ledger=True):
    """아르곤 한 번 호출. 반환 dict: ok, code(0/2/3/4/5), text, out, model, seconds, limited, note."""
    t0 = time.time()
    res = {"ok": False, "code": EXIT_FAIL, "text": "", "out": None, "model": model, "seconds": 0.0,
           "limited": False, "note": "", "agy": None}
    agy = find_agy()
    res["agy"] = agy
    if not agy:
        res.update(code=EXIT_USAGE, note="agy 실행 파일을 찾지 못함: " + " | ".join(c for c in _AGY_CANDIDATES if c))
        return res
    try:
        model, why = resolve_model(agy, model)
    except ValueError as e:
        res.update(code=EXIT_USAGE, note=str(e))
        return res
    res["model"] = model
    notes = [why] if why else []

    stamp = _stamp()
    work = os.path.join(WORK_ROOT, stamp)
    os.makedirs(work, exist_ok=True)
    out = out or os.path.join(work, "답.txt")
    os.makedirs(os.path.dirname(os.path.abspath(out)), exist_ok=True)
    res["out"] = out

    dirs = [os.path.abspath(d) for d in (add_dirs or []) if d]
    send = prompt
    if len(prompt) > LONG_PROMPT_CHARS:
        pf = os.path.join(work, "프롬프트.txt")
        open(pf, "w", encoding="utf-8").write(prompt)
        dirs.append(work)
        send = ("폴더에 있는 '프롬프트.txt' 파일을 처음부터 끝까지 읽고, 그 안의 지시를 그대로 수행하세요. "
                "답은 이 대화에 바로 쓰세요(파일에 쓰지 마세요).")
        notes.append("긴 프롬프트(%d자)는 파일로 전달" % len(prompt))
    dirs = list(dict.fromkeys(dirs))

    def build(m):
        cmd = [agy, "-p", send, "--model", m]
        for d in dirs:
            cmd += ["--add-dir", d]
        return cmd + ["--dangerously-skip-permissions", "--print-timeout", "%ds" % print_timeout]

    try:
        with _Slot(slot_wait, print_timeout + 600):
            rc, so, se, timed_out = _run_agy(build(model), print_timeout + 90)
            if (not timed_out and rc != 0 and model == DEFAULT_MODEL and not so.strip()
                    and _NOMODEL_RE.search(se + so) and not _LIMIT_RE.search(se + so)):
                notes.append("%s 호출 거절 -> %s로 한 번 재시도" % (model, FALLBACK_MODEL))
                model = res["model"] = FALLBACK_MODEL
                rc, so, se, timed_out = _run_agy(build(model), print_timeout + 90)
    except TimeoutError as e:
        res.update(code=EXIT_TIMEOUT, note=str(e), seconds=round(time.time() - t0, 1))
        return res

    text = so.strip()
    both = se + "\n" + so
    res["seconds"] = round(time.time() - t0, 1)
    if timed_out:
        res.update(code=EXIT_TIMEOUT)
        notes.append("시간 초과(%ds) - 이 호출만 끔" % (print_timeout + 90))
    elif (rc != 0 or len(text) < 1500) and _LIMIT_RE.search(both):
        res.update(code=EXIT_LIMIT, limited=True)
        notes.append("한도(429) 신호 - 이 계정·경로는 멈추고 전환")
    elif rc != 0 or not text:
        res.update(code=EXIT_FAIL)
        notes.append("종료코드 %s, 응답 %d자" % (rc, len(text)))
    else:
        res.update(ok=True, code=EXIT_OK)
    res["text"] = text
    res["note"] = "; ".join(n for n in notes if n)
    try:
        open(out, "w", encoding="utf-8").write(text + ("\n" if text else ""))
        meta = {k: res[k] for k in ("ok", "code", "model", "seconds", "limited", "note", "agy")}
        meta.update(add_dirs=dirs, rc=rc, route="agy(Antigravity) 아르곤 - 무료 경로, Vertex 아님",
                    stderr_tail=se.strip()[-500:], at=dt.datetime.now().isoformat(timespec="seconds"))
        json.dump(meta, open(out + ".meta.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    except OSError as e:
        res["note"] += "; 저장 실패: %s" % e
    if ledger:
        _ledger(model, res["ok"])
    return res


def main(argv=None):
    ap = argparse.ArgumentParser(description="agy 아르곤 호출 도우미 (사용법은 파일 맨 위 3줄)")
    ap.add_argument("--prompt", help="질문 본문")
    ap.add_argument("--prompt-file", help="질문을 담은 텍스트 파일(UTF-8)")
    ap.add_argument("--add-dir", action="append", default=[], help="agy가 읽고 쓸 폴더(여러 번 가능)")
    ap.add_argument("--model", default=DEFAULT_MODEL, help="기본 %s (없으면 medium)" % DEFAULT_MODEL)
    ap.add_argument("--out", help="답 저장 파일(기본 D:\\작업_대용량\\아르곤_호출\\<시각>\\답.txt)")
    ap.add_argument("--print-timeout", type=int, default=600, help="agy --print-timeout 초(기본 600)")
    ap.add_argument("--slot-wait", type=int, default=900, help="동시 3개 한도 자리 대기 초(기본 900)")
    ap.add_argument("--echo", action="store_true", help="답 전문을 화면에도 출력")
    ap.add_argument("--no-ledger", action="store_true", help="무료 사용량 장부에 적지 않음")
    ap.add_argument("--check", action="store_true", help="agy 경로·모델만 확인하고 끝냄")
    a = ap.parse_args(argv)
    try:
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
        sys.stderr.reconfigure(encoding="utf-8", errors="replace")
    except Exception:
        pass
    if a.check:
        agy = find_agy()
        if not agy:
            print("agy 없음", file=sys.stderr)
            return EXIT_USAGE
        try:
            m, why = resolve_model(agy, a.model)
        except ValueError as e:
            print(str(e), file=sys.stderr)
            return EXIT_USAGE
        print("agy=%s\n모델=%s %s" % (agy, m, why))
        return EXIT_OK
    prompt = a.prompt
    if a.prompt_file:
        prompt = open(a.prompt_file, encoding="utf-8-sig").read()
    if not prompt or not prompt.strip():
        print("--prompt 또는 --prompt-file 이 필요합니다", file=sys.stderr)
        return EXIT_USAGE
    r = call(prompt, add_dirs=a.add_dir, out=a.out, model=a.model, print_timeout=a.print_timeout,
             slot_wait=a.slot_wait, ledger=not a.no_ledger)
    label = {EXIT_OK: "성공", EXIT_USAGE: "사용법/환경 오류", EXIT_LIMIT: "한도(429)", EXIT_TIMEOUT: "시간초과", EXIT_FAIL: "실패"}
    print("[아르곤] %s 코드=%d 모델=%s %.0f초 저장=%s %s" % (label.get(r["code"], "?"), r["code"], r["model"], r["seconds"],
                                                       r["out"], r["note"]), file=sys.stderr)
    if a.echo and r["text"]:
        print(r["text"])
    elif r["ok"]:
        print(r["out"])
    return r["code"]


if __name__ == "__main__":
    sys.exit(main())
